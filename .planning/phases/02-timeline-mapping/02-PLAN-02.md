```c
/* ══════════════════════════════════════════════════════════════
   [T10] Timeline Round-Trip — face+zone+slot → DualFrame
   ══════════════════════════════════════════════════════════════ */
static void test_timeline_roundtrip(void) {
    printf("\n─── [T10] Timeline Round-Trip ───\n");

    /* Test 1: Known synthetic signature → verify DualFrame */
    int64_t vx = 100000;
    int64_t vy = 50000;

    TWFaceCapture best;
    int n_faces = tw_capture_priority(vx, vy, &best, 7);
    ASSERT(n_faces > 0);
    ASSERT(best.face < 12);
    ASSERT(best.tring_pos < 1440);

    /* Map to DualFrame via geo_frame_seek.h */
    DualFrame df = frame_at(best.tring_pos);

    /* Verify round-trip: enc == tring_pos */
    ASSERT_EQ(df.enc, best.tring_pos);

    /* verify DualFrame face matches TW capture face */
    ASSERT_EQ(df.face, best.face);

    /* verify DualFrame slot matches TW capture zone*6+slot */
    uint8_t expected_slot = (uint8_t)(best.tring_pos % 120);
    ASSERT_EQ(df.slot, expected_slot);

    /* verify DualFrame fields are in valid range */
    ASSERT(df.ico_idx < 162);
    ASSERT(df.phase < 12);
    ASSERT(df.h.group < 3);
    ASSERT(df.h.edge < 3);
    ASSERT(df.p.step < 4);
    ASSERT(df.p.sub < 3);

    printf("  sig(%d,%d) → face=%d slot=%d tring=%d → enc=%d face=%d slot=%d\n",
           (int)vx, (int)vy, best.face, best.slot, best.tring_pos,
           df.enc, df.face, df.slot);

    /* Test 2: Determinism — same inputs → same DualFrame */
    TWFaceCapture best2;
    tw_capture_priority(vx, vy, &best2, 7);
    DualFrame df2 = frame_at(best2.tring_pos);
    ASSERT_EQ(df.enc, df2.enc);
    ASSERT_EQ(df.face, df2.face);
    ASSERT_EQ(df.slot, df2.slot);
    ASSERT_EQ(df.ico_idx, df2.ico_idx);
    ASSERT_EQ(df.phase, df2.phase);

    /* Test 3: Multiple signatures produce valid mappings */
    int64_t test_sigs[][2] = {
        {0, 0},
        {207360, 0},
        {0, 207360},
        {-100000, 50000},
        {50000, -100000},
    };
    int n_sigs = sizeof(test_sigs) / sizeof(test_sigs[0]);
    for (int i = 0; i < n_sigs; i++) {
        TWFaceCapture c;
        tw_capture_priority(test_sigs[i][0], test_sigs[i][1], &c, 7);
        DualFrame d = frame_at(c.tring_pos);
        ASSERT(d.enc == c.tring_pos);
        ASSERT(d.face < 12);
        ASSERT(d.slot < 120);
        ASSERT(d.ico_idx < 162);
        ASSERT(d.phase < 12);
        printf("  sig[%d] → tring=%d enc=%d face=%d slot=%d\n",
               i, c.tring_pos, d.enc, d.face, d.slot);
    }

    /* Test 4: World B mapping via cpair */
    for (uint16_t tp = 0; tp < 120; tp += 10) {
        uint16_t wb = frame_cpair(tp);
        ASSERT(wb >= 720 && wb < 1440);
        DualFrame df_wb = frame_at(wb);
        ASSERT(df_wb.enc == wb);
        /* World B face = (wb / 120) — should be opposite pole */
        ASSERT(df_wb.face == (uint8_t)(wb / 120));
    }

    CHECK(g_pass > 0, "timeline round-trip: all checks passed");
}
```
