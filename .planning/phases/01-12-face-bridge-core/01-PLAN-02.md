```c
static void test_orchestrator(void) {
    printf("\n─── [T9] Orchestrator Unit Test ───\n");

    /* Test init/free */
    TWCapture12FaceResult r;
    tw_capture_12face_init(&r);
    ASSERT_EQ(r.n_tensors, 0u);

    /* Create a minimal TWCapture12FaceResult by simulating priority capture */
    tw_capture_12face_init(&r);

    /* Manually capture a face to verify rewind integration */
    int64_t vx = 100000;
    int64_t vy = 50000;
    TWFaceCapture best;
    int n_faces = tw_capture_priority(vx, vy, &best, 7);
    ASSERT(n_faces > 0);
    ASSERT(best.face < 12);
    ASSERT(best.tring_pos < 1440);

    tw_face_rewind_store(&r.rewind, &best);
    ASSERT(tw_rewind_occupied(&r.rewind) > 0);
    ASSERT(tw_rewind_has(&r.rewind, best.tring_pos));

    r.best = best;
    r.n_tensors = 1;

    /* Verify freeze logic */
    if (tw_face_is_frozen(&best, 12)) {
        ASSERT(r.n_frozen < r.freeze_cap || r.freeze_cap == 0);
    }

    tw_capture_12face_free(&r);
```

```c
/* ══════════════════════════════════════════════════════════════
   [B1] 12-Face Benchmark
   ══════════════════════════════════════════════════════════════ */
#ifdef BENCHMARK
#include <time.h>

static void benchmark_12face(const char *tensors_dir) {
    printf("\n─── [B1] 12-Face Benchmark ───\n");

    RawBridge rb;
    memset(&rb, 0, sizeof(rb));
    if (rb_load(&rb, tensors_dir) != RB_OK) {
        printf("  SKIP: no tensor data at %s\n", tensors_dir);
        g_pass++;
        return;
    }

    /* Count tensors available */
    uint32_t n_avail = 0;
    for (uint32_t i = 0; i < RB_MAX_ENTRIES; i++)
        if (rb.entries[i].occupied) n_avail++;

    if (n_avail == 0) { rb_free(&rb); g_pass++; return; }

    /* Warmup: run once */
    TWCapture12FaceResult warmup;
    tw_capture_12face_init(&warmup);
    for (uint32_t i = 0; i < RB_MAX_ENTRIES; i++) {
        if (!rb.entries[i].occupied) continue;
        tw_capture_tensor_12face(&rb, rb.entries[i].name, 12, 0, &warmup);
    }
    tw_capture_12face_free(&warmup);

    /* Timed run */
    clock_t start = clock();
    int iterations = 3;
    for (int iter = 0; iter < iterations; iter++) {
        TWCapture12FaceResult result;
        tw_capture_12face_init(&result);
        for (uint32_t i = 0; i < RB_MAX_ENTRIES; i++) {
            if (!rb.entries[i].occupied) continue;
            tw_capture_tensor_12face(&rb, rb.entries[i].name, 12, 0, &result);
        }
        tw_capture_12face_free(&result);
    }
    clock_t end = clock();

    double elapsed_s = (double)(end - start) / CLOCKS_PER_SEC;
    double total_tensors = (double)n_avail * iterations;
    double tps = total_tensors / elapsed_s;

    printf("  Tensors: %.0f in %.3fs = %.0f t/s\n", total_tensors, elapsed_s, tps);
    printf("  Target: 2000 t/s (single-face baseline: 26000 t/s)\n");

    ASSERT(tps > 100.0); /* minimum: must be >100 t/s */
    if (tps >= 2000.0) {
        printf("  ✓ MEETS TARGET (>= 2000 t/s)\n");
    } else {
        printf("  ⚠ Below target: %.0f < 2000 t/s\n", tps);
    }

    rb_free(&rb);
    CHECK(g_pass > 0, "benchmark: completed");
}
#endif /* BENCHMARK */
```
